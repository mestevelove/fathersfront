<script>
(function () {
  var form = document.querySelector(".ff-dispatch-optin__form");
  var response = document.querySelector(".ff-dispatch-optin__response");
  if (!form || !response) return;

  var button = form.querySelector('button[type="submit"]');
  var submitted = false;

  form.addEventListener("submit", function () {
    submitted = true;
    button.disabled = true;
    button.textContent = "Sending...";
  });

  response.addEventListener("load", function () {
    if (!submitted) return;

    try {
      if (!response.contentWindow.location.href ||
          response.contentWindow.location.href === "about:blank") return;
    } catch (error) {
      /* A cross-origin response means Systeme.io answered. */
    }

    button.textContent = "Received — Welcome to the Front";
    form.querySelectorAll("input").forEach(function (input) {
      input.disabled = true;
    });
  });
})();
</script>
