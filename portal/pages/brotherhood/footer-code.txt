<!--
  FATHERS FRONT - SHARED PORTAL JAVASCRIPT
  Install on every Fathers Front Systeme.io page.
  Controls the mobile menu, footer year, and campaign-parameter forwarding.
-->
<script>
(function () {
  function closeMenu(header) {
    if (!header) return;
    var toggle = header.querySelector(".ff-sio-header__toggle");
    var nav = header.querySelector(".ff-sio-header__nav");
    if (!toggle || !nav) return;
    toggle.setAttribute("aria-expanded", "false");
    toggle.setAttribute("aria-label", "Open navigation");
    nav.classList.remove("is-open");
    document.documentElement.style.overflow = "";
  }

  if (!window.__ffPortalMenuBound) {
    window.__ffPortalMenuBound = true;

    document.addEventListener("click", function (event) {
      var toggle = event.target.closest(".ff-sio-header__toggle");

      if (toggle) {
        var header = toggle.closest(".ff-sio-header");
        var nav = header && header.querySelector(".ff-sio-header__nav");
        if (!nav) return;

        var opening = toggle.getAttribute("aria-expanded") !== "true";
        toggle.setAttribute("aria-expanded", String(opening));
        toggle.setAttribute("aria-label", opening ? "Close navigation" : "Open navigation");
        nav.classList.toggle("is-open", opening);
        document.documentElement.style.overflow = opening ? "hidden" : "";
        return;
      }

      var navLink = event.target.closest(".ff-sio-header__nav a");
      if (navLink) closeMenu(navLink.closest(".ff-sio-header"));
    });

    document.addEventListener("keydown", function (event) {
      if (event.key === "Escape") {
        document.querySelectorAll(".ff-sio-header").forEach(closeMenu);
      }
    });
  }

  /*
   * Systeme.io renders checkout-page footer markup after this script runs.
   * Listen at the document level so the Dispatch form works whether it is
   * present immediately or inserted later inside the checkout structure.
   */
  if (!window.__ffPortalDispatchBound) {
    window.__ffPortalDispatchBound = true;

    document.addEventListener("submit", async function (event) {
      var dispatchForm = event.target.closest("[data-ff-dispatch-form]");
      if (!dispatchForm) return;

      event.preventDefault();

      var footer = dispatchForm.closest(".ff-sio-footer");
      var button = dispatchForm.querySelector('button[type="submit"]');
      var status = footer && footer.querySelector("[data-ff-dispatch-status]");
      var fields = dispatchForm.querySelectorAll("input");
      var formData = new FormData(dispatchForm);

      button.disabled = true;
      button.textContent = "Sending...";
      if (status) status.textContent = "";

      try {
        var response = await fetch("https://fathersfront.com/api/dispatch", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            firstName: formData.get("firstName") || "",
            email: formData.get("email") || "",
            website: formData.get("website") || ""
          })
        });

        if (!response.ok) throw new Error("Dispatch signup failed");

        fields.forEach(function (field) {
          if (field.name !== "website") field.value = "";
          field.disabled = true;
        });
        button.textContent = "Received - Welcome to the Front";
      } catch (error) {
        button.disabled = false;
        button.textContent = "Join the Dispatch";
        if (status) status.textContent = "Something went wrong. Please try again.";
      }
    });
  }

  function initializeFooter() {
    var footer = document.querySelector(".ff-sio-footer");
    if (!footer || footer.dataset.ffInitialized) return;
    footer.dataset.ffInitialized = "true";

    var year = footer.querySelector("[data-ff-current-year]");
    if (year) year.textContent = new Date().getFullYear();

    var allowed = ["utm_source", "utm_medium", "utm_campaign", "utm_content", "utm_term"];
    var incoming = new URLSearchParams(window.location.search);
    var campaign = new URLSearchParams();

    allowed.forEach(function (key) {
      if (incoming.has(key)) campaign.set(key, incoming.get(key));
    });

    if (!campaign.toString()) return;

    footer.querySelectorAll('a[href^="https://fathersfront.com/"]').forEach(function (link) {
      var url = new URL(link.href);
      campaign.forEach(function (value, key) {
        if (!url.searchParams.has(key)) url.searchParams.set(key, value);
      });
      link.href = url.toString();
    });
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", initializeFooter, { once: true });
  } else {
    initializeFooter();
  }
})();
</script>
