<script>
(function () {
  document.querySelectorAll(".ff-systeme-capture").forEach(function (capture) {
    if (capture.dataset.ffCaptureReady === "true") return;
    capture.dataset.ffCaptureReady = "true";

    var form = capture.querySelector(".ff-systeme-capture__form");
    var button = form.querySelector("button[type='submit']");
    var status = capture.querySelector(".ff-systeme-capture__status");

    form.addEventListener("submit", async function (event) {
      event.preventDefault();
      var data = new FormData(form);

      button.disabled = true;
      button.textContent = "Sending...";
      status.textContent = "";
      status.classList.remove("is-error");

      try {
        var response = await fetch("https://fathersfront.com/api/dispatch", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            firstName: data.get("firstName"),
            email: data.get("email"),
            website: data.get("website"),
            tag: capture.dataset.ffCaptureTag
          })
        });

        if (!response.ok) throw new Error("Submission failed");
        form.reset();
        form.querySelectorAll("input").forEach(function (input) {
          input.disabled = true;
        });
        button.textContent = button.dataset.successLabel;
      } catch (error) {
        button.disabled = false;
        button.textContent = button.dataset.idleLabel;
        status.textContent = "Something went wrong. Please try again.";
        status.classList.add("is-error");
      }
    });
  });
})();
</script>
